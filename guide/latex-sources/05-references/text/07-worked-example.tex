\begin{frame}[fragile]{Complete reference example: the starboard rotor}
\small Continue the same file. The two diameters differ on purpose: one
advance ratio written once in a row's flight condition resolves against
each rotor's own diameter, so unlike rotors turn at unlike speeds from
one written number.
\begin{lstlisting}[basicstyle=\ttfamily\scriptsize,numbers=none]
[STARBOARD]
kind = "rotor"
x_m = 4.5
y_m = -2.5
z_m = 0.0
axis = "X"
rpm_sign = -1        # counter-rotating
diameter_m = 1.8288
families_general = []
families_blades = ["Blade2"]
blade1 = { azimuth_deg = 0.0, zero = "Y" }
\end{lstlisting}
\setupsources{\code{cases/\_\_init\_\_.py}; \code{cases/workflows.py};

\end{frame}
