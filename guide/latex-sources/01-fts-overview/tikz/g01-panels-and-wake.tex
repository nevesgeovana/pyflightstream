% A half wing seen from above: flow-aligned quadrilateral faces, the marked
% trailing edge, and the wake filaments it sheds. An unmarked edge sheds none.
% Shared fragment: no sectioning, no frame, no size command, no float. Wrapped by the caller.
\begin{tikzpicture}[scale=0.9, font=\scriptsize]
% planform corners: root LE (0,0), root TE (1.6,0), tip LE (0.7,3.0), tip TE (1.6,3.0)
\def\ns{6}
\def\nc{5}
\foreach \j in {0,...,\ns}{
  \pgfmathsetmacro{\y}{3.0*\j/\ns}
  \pgfmathsetmacro{\xle}{0.7*\j/\ns}
  \draw[myblue!60] (\xle,\y) -- (1.6,\y);}
\foreach \i in {0,...,\nc}{
  \pgfmathsetmacro{\f}{\i/\nc}
  \draw[myblue!60] ({1.6*\f},0) -- ({0.7+0.9*\f},3.0);}
\draw[thick, myblue] (0,0) -- (0.7,3.0) -- (1.6,3.0) -- (1.6,0) -- cycle;
% the marked trailing edge
\draw[ultra thick, darkred] (1.6,0) -- (1.6,3.0);
% wake filaments from each trailing-edge node
\foreach \j in {0,...,\ns}{
  \pgfmathsetmacro{\y}{3.0*\j/\ns}
  \draw[dashed, darkred!70] (1.6,\y) -- (4.2,\y);}
\draw[-{Stealth}] (-0.9,1.5) -- (-0.2,1.5) node[midway, above] {$V_\infty$};
\node[darkred, align=left, anchor=west] at (1.75,3.35) {marked trailing edge};
\node[darkred!80, align=center] at (3.0,-0.35) {wake filaments};
\node[myblue, align=center] at (0.45,-0.45) {faces};
\draw[gray, dashed] (-0.3,0) -- (4.2,0);
\node[gray, anchor=west] at (2.3,0.18) {\tiny symmetry plane};
\end{tikzpicture}
