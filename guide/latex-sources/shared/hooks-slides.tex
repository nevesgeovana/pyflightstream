% ---------------------------------------------------- the slides' own tools --
% The PDF's Title, stated in each deck's PREAMBLE:
%   \guidepdftitle{<number>}{<short title for the footer>}
% Beamer writes pdftitle from \title as the document begins (usepdftitle, its
% default), and hyperref fixes the PDF's information there, so a Title set in
% the body is never written: all eight decks of 0.30.0 carried an empty Title
% from a \hypersetup inside the title frame. The title frame checks that its
% number and short title are the ones stated here.
\newcommand{\guidepdfdeck}{}
\newcommand{\guidepdftitle}[2]{%
  \gdef\guidepdfdeck{#1: #2}%
  \title{Guide #1: #2, \pkgname}}

% The title frame of every deck.
%   \guidetitleframe{<number>}{<title>}{<short title for the footer>}{<subtitle>}{<sources line>}
% It prints the one statement of the version the deck is written for and the
% full authorship and licence notice.
\newcommand{\guidetitleframe}[5]{%
  \gdef\guidenumber{#1}\gdef\guideshort{#3}%
  \def\guidetitledeck{#1: #3}%
  \ifx\guidetitledeck\guidepdfdeck\else
    \GenericError{}{Guide deck: the preamble's \string\guidepdftitle\space
      states `\guidepdfdeck', the title frame `#1: #3'}{}{State the same
      number and short title in both.}%
  \fi
  \begin{frame}[plain,noframenumbering]
    \vspace*{4mm}
    {\color{SoftGray}\small\bfseries \guideseries, guide #1 of \guidecount, numbered \guiderange\par}
    \vspace{5mm}
    {\color{hanblue}\huge\bfseries Guide #1: #2\par}
    \vspace{2mm}
    {\large #4\par}
    \vspace{7mm}
    {\small \guidewritten.\par}
    \vspace{1mm}
    {\small #5\par}
    \vfill
    {\scriptsize\color{SoftGray} \pkgname{} is the work of \pkgauthor:
    \url{\pkgrepo}, concept DOI \texttt{\pkgdoiconcept}.\par
    \guidenotice\par}
  \end{frame}}

% A guide figure fitted to the frame: never wider than the text, never taller
% than the given fraction of the frame.
\newcommand{\slidefig}[2][0.72]{%
  \begin{center}
  \adjustbox{max width=\linewidth, max totalheight=#1\textheight}{\input{tikz/#2.tex}}%
  \end{center}}

% A value each site supplies for itself. Grey and italic, impossible to
% mistake for content.
\newcommand{\tofill}[1]{{\color{SoftGray}\itshape\small [your site: #1]}}

% The numbered references of a deck, cited in the text as [1], [2], ...
% and listed on the deck's last frame, as in a paper. Each entry is a
% \bibitem with a key, so the entries are a bibliography a reader or a tool
% can count; the key is not printed and the number is the list's:
%   \begin{frame}{References}\begin{reflist}\bibitem{key} ... \end{reflist}\end{frame}
\newenvironment{reflist}
  {\scriptsize\setbeamertemplate{enumerate item}{[\insertenumlabel]}%
   \renewcommand{\bibitem}[1]{\item}%
   \begin{enumerate}\setlength{\itemsep}{2pt}}
  {\end{enumerate}}
